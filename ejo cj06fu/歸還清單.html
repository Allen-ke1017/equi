function doGet(e) { // 1. 開啟試算表與指定的「設備清單」工作表 const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("設備清單"); if (!sheet) { return ContentService.createTextOutput("❌ 找不到「設備清單」工作表！"); } const data = sheet.getDataRange().getValues(); // 取得所有資料 // 假設試算表結構如下： // 欄 A (index 0): 設備編號 // 欄 B (index 1): 設備名稱 // 欄 C (index 2): 目前狀態 ("可借用" 或 "已借出") // 欄 D (index 3): 最後異動時間 let returnedList = []; // 2. 從第 2 列開始尋找所有狀態為「可借用」的設備 for (let i = 1; i < data.length; i++) { const id = data[i][0]; const name = data[i][1]; const status = data[i][2]; const lastUpdate = data[i][3]; if (status === "可借用") { // 格式化時間（若有記錄） let timeStr = "無紀錄"; if (lastUpdate) { timeStr = lastUpdate instanceof Date ? Utilities.formatDate(lastUpdate, "GMT+8", "yyyy-MM-dd HH:mm:ss") : lastUpdate; } returnedList.push(`• [${id}] ${name} (歸還時間: ${timeStr})`); } } // 3. 組合回應訊息 if (returnedList.length === 0) { return ContentService.createTextOutput("📋 目前沒有已歸還（可借用）的設備。"); } const resultText = `📋 已歸還（可借用）設備清單（共 ${returnedList.length} 件）：\n` + "-----------------------------------\n" + returnedList.join("\n"); return ContentService.createTextOutput(resultText); }
