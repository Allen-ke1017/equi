function doGet(e) { var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); // ================= 1. 註冊新帳號 ================= if (action === "register") { var userSheet = ss.getSheetByName("Users"); // 請確認工作表分頁名稱是否為 Users if (!userSheet) return ContentService.createTextOutput("錯誤：找不到 Users 工作表"); var userId = String(e.parameter.userId || "").trim(); var userName = String(e.parameter.userName || "").trim(); var password = String(e.parameter.password || "").trim(); if (!userId || !password || !userName) { return ContentService.createTextOutput("註冊失敗：資料不完整"); } // 檢查 A 欄 (user_id) 是否已被註冊 var data = userSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === userId) { return ContentService.createTextOutput("註冊失敗：此帳號已經存在！"); } } // 依照您的 5 個欄位順序寫入：user_id, name, role, status, password // 預設 role 為 "user"，status 為 "active"（若您希望註冊後需管理員審核才開通，可將 "active" 改為 "pending"） userSheet.appendRow(["'" + userId, userName, "user", "active", "'" + password]); return ContentService.createTextOutput("註冊成功"); } // ================= 2. 帳號密碼登入驗證 ================= if (action === "login") { var userSheet = ss.getSheetByName("Users"); if (!userSheet) return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "找不到 Users 工作表" })); var userId = String(e.parameter.userId || "").trim(); var password = String(e.parameter.password || "").trim(); var data = userSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var rowId = String(data[i][0]).trim(); // A欄: user_id var rowName = String(data[i][1]).trim(); // B欄: name var rowRole = String(data[i][2]).trim(); // C欄: role var rowStatus = String(data[i][3]).trim(); // D欄: status var rowPwd = String(data[i][4]).trim(); // E欄: password if (rowId === userId && rowPwd === password) { // 檢查 D 欄 status 狀態（若為停用/未開通則阻擋登入） if (rowStatus.toLowerCase() === "disabled" || rowStatus === "停用" || rowStatus.toLowerCase() === "pending") { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "此帳號目前無法使用（狀態：" + rowStatus + "），請聯繫管理員！" })); } // 驗證成功，回傳 userId、userName 與 role return ContentService.createTextOutput(JSON.stringify({ status: "success", userId: rowId, userName: rowName, role: rowRole })); } } return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "帳號或密碼錯誤！" })); } // ================= 3. 原本的設備掃描借還邏輯 ================= // （請將您原本寫在 doGet 裡面的掃描借還程式碼保留在這裡） var id = e.parameter.id; var borrowerId = e.parameter.userId; var borrowerName = e.parameter.userName; // ... 您原本的程式碼 ... }
